# AEDP3消融实验多终端运行命令
# 支持注意力通道消融和编码器融合策略消融
# 注意：需要先启动GS2服务器（见下方终端2-5）

# 终端1: 生成adroit数据集（如果还没有生成）
export CUDA_VISIBLE_DEVICES=0
export GS2_CONDA_ENV=aedp3_vis
export DEVICE=cuda:0
bash scripts/make_adroit_datasets.sh

# 终端2: GPU0的GS2服务器 (端口5000) - 用于有attn任务
cd Grounded-SAM-2
CUDA_VISIBLE_DEVICES=0 python gs2_api_server.py --device cuda:0 --port 5000

# 终端3: GPU1的GS2服务器 (端口5001) - 用于有attn任务
cd Grounded-SAM-2
CUDA_VISIBLE_DEVICES=1 python gs2_api_server.py --device cuda:1 --port 5001

# 终端4: GPU2的GS2服务器 (端口5002) - 用于有attn任务
cd Grounded-SAM-2
CUDA_VISIBLE_DEVICES=2 python gs2_api_server.py --device cuda:2 --port 5002

# 终端5: GPU3的GS2服务器 (端口5003) - 用于有attn任务
cd Grounded-SAM-2
CUDA_VISIBLE_DEVICES=3 python gs2_api_server.py --device cuda:3 --port 5003

# ==================== 消融实验训练任务 ====================

# 终端6: GPU0训练所有变体 - adroit_pen任务
export CUDA_VISIBLE_DEVICES=0
export GPU_ID=0
export SEED=0
export CONFIG_NAME=dp3
export GS2_PORT=5000
export TASKS="adroit_pen"
export ABLATION_VARIANTS="DP3,AEDP3-Full,AEDP3-C0,AEDP3-C1,AEDP3-C2,AEDP3-C01,AEDP3-C02,AEDP3-C12,AEDP3-Late,AEDP3-Early"
bash scripts/run_ablation_experiments.sh

# 终端7: GPU1训练所有变体 - adroit_door任务
export CUDA_VISIBLE_DEVICES=1
export GPU_ID=1
export SEED=0
export CONFIG_NAME=dp3
export GS2_PORT=5001
export TASKS="adroit_door"
